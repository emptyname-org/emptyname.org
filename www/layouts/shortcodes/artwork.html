{{- /* Embed an artwork as an inline illustration in any page. Usage: {{< artwork id="example-work" >}} small floated thumbnail {{< artwork id="example-work" side="left" >}} side = right (default) | left | center {{< artwork id="example-work" size="full" >}} big block image, full content width {{< artwork id="example-work" size="square" >}} square thumbnail Captioned with the artwork's title (linked to its page); click → lightbox. All metadata comes from the artwork's page (data/artworks.json) - defined once, reused here. */ -}} {{- $id := .Get "id" -}} {{- $side := .Get "side" | default "right" -}} {{- $size := .Get "size" -}} {{- $big := in (slice "full" "big") $size -}} {{- $square := eq $size "square" -}} {{- with site.GetPage (printf "/artwork/%s" $id) -}} {{- if $square -}} {{- else -}} {{- $img := index .Params.images 0 -}} {{- $first := index .Params.media 0 -}} {{- $alt := $first.alt | default .Title -}} {{- $optimized := partial "boundary-image.html" (dict "src" $img "displayWidth" (cond $big 1200 600)) -}} {{- end -}} {{- else -}} {{- warnf "artwork shortcode: no artwork %q (referenced in %s)" $id .Page.File.Path -}} {{- end -}}
